This survey is delivered as a QSF file — Qualtrics Survey Format, a single file holding an entire survey, exported from one Qualtrics account and imported into another. Importing it rebuilds every block, question and flow element in one step, so nothing has to be assembled by hand. The steps below are the order to work in; the code boxes throughout Part 2 are there to verify the import and to repair anything that did not survive it.
  1. Check two account permissions
    Allow JavaScript and Allow All HTML Markup, under account settings. Do this before anything else. Without them the import fails when the file carries custom code, or — worse — it succeeds and then renders blank for participants while still looking correct in the editor.
  2. Import the QSF file into a throwaway draft survey first
    That import is the real test, and it is the whole build in one step: it creates every block, every question, the Survey Flow, and the embedded-data declarations.  * qualtrics_api.js — the ONLY place the task engines touch Qualtrics.
 *
 * Why this file exists
 * --------------------
 * The embedded-data setters and the page-submit hook live on the STATIC
 * `Qualtrics.SurveyEngine` namespace, NOT on the question instance that
 * `this` refers to inside addOnload/addOnReady. Calling them on the instance
 * (`q.setJSEmbeddedData(...)`) throws, and every recorded value is lost — the
 * task renders and plays perfectly, then writes nothing. Guarding the call
 * (`if (q.addOnPageSubmit) ...`) makes that failure silent, which is worse.
 *
 * So: engines call QAPI.setED / QAPI.onSubmit. Those resolve the real static
 * API at call time and THROW LOUDLY if it is not there. A missing API must
 * break the local harness and the Qualtrics preview, not quietly drop data.
 *
 * Layout compatibility
 * --------------------
 * `setJSEmbeddedData` is the current setter and the only one that works in the
 * New Survey Taking Experience; `setEmbeddedData` is deprecated but still works
 * in the legacy layout. We prefer the former and fall back to the latter, so
 * the same code is correct under either layout.
 *
 * Values are coerced to String, and null/undefined become "" rather than the
 * literal text "null"/"undefined".
 * ==========================================================================*/

/* ============================================================================
 * qualtrics_api.js — the ONLY place the task engines touch Qualtrics.
 *
 * Why this file exists
 * --------------------
 * The embedded-data setters and the page-submit hook live on the STATIC
 * `Qualtrics.SurveyEngine` namespace, NOT on the question instance that
 * `this` refers to inside addOnload/addOnReady. Calling them on the instance
 * (`q.setJSEmbeddedData(...)`) throws, and every recorded value is lost — the
 * task renders and plays perfectly, then writes nothing. Guarding the call
 * (`if (q.addOnPageSubmit) ...`) makes that failure silent, which is worse.
 *
 * So: engines call QAPI.setED / QAPI.onSubmit. Those resolve the real static
 * API at call time and THROW LOUDLY if it is not there. A missing API must
 * break the local harness and the Qualtrics preview, not quietly drop data.
 *
 * Layout compatibility
 * --------------------
 * `setJSEmbeddedData` is the current setter and the only one that works in the
 * New Survey Taking Experience; `setEmbeddedData` is deprecated but still works
 * in the legacy layout. We prefer the former and fall back to the latter, so
 * the same code is correct under either layout.
 *
 * Values are coerced to String, and null/undefined become "" rather than the
 * literal text "null"/"undefined".
 * ==========================================================================*/
(function (root) {
  "use strict";

  function engine() {
    return root.Qualtrics && root.Qualtrics.SurveyEngine;
  }

  /* Write one embedded-data field. Throws if no setter is available. */
  function setED(name, value) {
    var E = engine();
    if (!E) {
      throw new Error("QAPI.setED(" + name + "): Qualtrics.SurveyEngine is not available");
    }
    var v = (value === null || value === undefined) ? "" : String(value);
    if (typeof E.setJSEmbeddedData === "function") {
      E.setJSEmbeddedData(name, v);
      return;
    }
    if (typeof E.setEmbeddedData === "function") {
      E.setEmbeddedData(name, v);
      return;
    }
    throw new Error("QAPI.setED(" + name + "): no embedded-data setter on " +
                    "Qualtrics.SurveyEngine — the value would have been lost");
  }

  /* Read one embedded-data field back, with a fallback. Never pipe ${e://...}
   * into JS to do this: piped text is resolved server-side before the page is
   * sent, so it is stale, and a quote in the value is a syntax error. */
  function getED(name, fallback) {
    var E = engine();
    var raw = (E && typeof E.getJSEmbeddedData === "function")
      ? E.getJSEmbeddedData(name) : null;
    return (raw === null || raw === undefined || raw === "") ? fallback : raw;
  }

  /* Register a page-submit handler — where all data writes belong. Data written
   * in addOnUnload races the page POST and is routinely lost. */
  function onSubmit(fn) {
    var E = engine();
    if (!E || typeof E.addOnPageSubmit !== "function") {
      throw new Error("QAPI.onSubmit: Qualtrics.SurveyEngine.addOnPageSubmit is " +
                      "unavailable — results would never be written");
    }
    E.addOnPageSubmit(fn);
  }

  root.QAPI = { setED: setED, getED: getED, onSubmit: onSubmit };
})(typeof window !== "undefined" ? window : this);

/* ============================================================================
 * rotation.js — mental-rotation task (same-vs-mirror judgement).
 * ----------------------------------------------------------------------------
 * Each trial shows two shapes side by side. The right shape is the left shape
 * either ROTATED (a "same" trial) or MIRRORED and rotated (a "mirror" trial).
 * The participant clicks "Same" or "Mirror." As many as possible in a fixed
 * time; score = number correct (we also record attempts).
 *
 * Stimuli are random polyominoes generated on the fly (license-clean, no
 * external assets, fully self-contained). Reference shapes are constrained to
 * be CHIRAL — i.e. their mirror image is not equal to any of their own
 * rotations — so "same" vs "mirror" is always unambiguous. "Same" trials use a
 * non-zero rotation, so a genuine mental rotation is required.
 *
 * Note: this is a 2-D polyomino implementation. If a stronger, more classic
 * 3-D Shepard-Metzler male-advantage stimulus is preferred, that is a separate
 * stimulus generator; the task scaffolding here (timer, scoring, fields) is
 * reusable either way.
 *
 * Pure logic (normalize / rotate / mirror / chirality / makeTrial) takes an
 * injectable RNG and touches no DOM; DOM wiring is in run().
 * ==========================================================================*/
(function (root, factory) {
  if (typeof module === "object" && module.exports) module.exports = factory();
  else root.Rotation = factory();

  "use strict";

  function normalize(cells) {
    var minR = Infinity, minC = Infinity;
    cells.forEach(function (p) { if (p[0] < minR) minR = p[0]; if (p[1] < minC) minC = p[1]; });
    return cells.map(function (p) { return [p[0] - minR, p[1] - minC]; })
      .sort(function (a, b) { return a[0] - b[0] || a[1] - b[1]; });
  }
  function key(cells) { return normalize(cells).map(function (p) { return p[0] + ":" + p[1]; }).join(","); }
  // 90-degree rotation: (r,c) -> (c, -r), then renormalize.
  function rotate(cells) { return normalize(cells.map(function (p) { return [p[1], -p[0]]; })); }
  // reflection across a vertical axis: (r,c) -> (r, -c).
  function mirror(cells) { return normalize(cells.map(function (p) { return [p[0], -p[1]]; })); }

  function rotations(cells) {
    var out = [cells], cur = cells;
    for (var i = 0; i < 3; i++) { cur = rotate(cur); out.push(cur); }
    return out;
  }
  // chiral = mirror image is not congruent to any rotation of the shape.
  function isChiral(cells) {
    var rotKeys = {};
    rotations(cells).forEach(function (r) { rotKeys[key(r)] = true; });
    return !rotKeys[key(mirror(cells))];
  }

  // Random polyomino of `size` connected cells via random growth.
  function genPolyomino(size, rng) {
    rng = rng || Math.random;
    var occ = { "0:0": true }, cells = [[0, 0]];
    var DIRS = [[1, 0], [-1, 0], [0, 1], [0, -1]];
    var guard = 0;
    while (cells.length < size && guard++ < 5000) {
      var base = cells[Math.floor(rng() * cells.length)];
      var d = DIRS[Math.floor(rng() * 4)];
      var nr = base[0] + d[0], nc = base[1] + d[1], k = nr + ":" + nc;
      if (!occ[k]) { occ[k] = true; cells.push([nr, nc]); }
    }
    return normalize(cells);
  }
  function genChiral(size, rng) {
    for (var i = 0; i < 200; i++) {
      var p = genPolyomino(size, rng);
      if (isChiral(p)) return p;
    }
    return genPolyomino(size, rng); // fallback (rare)
  }

  // makeTrial -> { reference, comparison, answer:"same"|"mirror" }
  function makeTrial(size, rng, forceSame) {
    rng = rng || Math.random;
    var ref = genChiral(size, rng);
    var same = (typeof forceSame === "boolean") ? forceSame : rng() < 0.5;
    var base = same ? ref : mirror(ref);

    // trials were allowed k = 0, a quarter of them showed the comparison as a pure
    // upright reflection, so "looks flipped but still upright => Mirror" was a
    // shortcut that never misfired: free points with no mental rotation, and
    // unequal difficulty between the two response categories.
    var k = 1 + Math.floor(rng() * 3);
    var comp = base;
    for (var i = 0; i < k; i++) comp = rotate(comp);
    return { reference: ref, comparison: comp, answer: same ? "same" : "mirror" };
  }

    genChiral: genChiral, makeTrial: makeTrial, run: run
  };

/* ============================================================================
 * qualtrics_api.js — the ONLY place the task engines touch Qualtrics.
 *
 * Why this file exists
 * --------------------
 * The embedded-data setters and the page-submit hook live on the STATIC
 * `Qualtrics.SurveyEngine` namespace, NOT on the question instance that
 * `this` refers to inside addOnload/addOnReady. Calling them on the instance
 * (`q.setJSEmbeddedData(...)`) throws, and every recorded value is lost — the
 * task renders and plays perfectly, then writes nothing. Guarding the call
 * (`if (q.addOnPageSubmit) ...`) makes that failure silent, which is worse.
 *
 * So: engines call QAPI.setED / QAPI.onSubmit. Those resolve the real static
 * API at call time and THROW LOUDLY if it is not there. A missing API must
 * break the local harness and the Qualtrics preview, not quietly drop data.
 *
 * Layout compatibility
 * --------------------
 * `setJSEmbeddedData` is the current setter and the only one that works in the
 * New Survey Taking Experience; `setEmbeddedData` is deprecated but still works
 * in the legacy layout. We prefer the former and fall back to the latter, so
 * the same code is correct under either layout.
 *
 * Values are coerced to String, and null/undefined become "" rather than the
 * literal text "null"/"undefined".
 * ==========================================================================*/
(function (root) {
  "use strict";

  function engine() {
    return root.Qualtrics && root.Qualtrics.SurveyEngine;
  }

  /* Write one embedded-data field. Throws if no setter is available. */
  function setED(name, value) {
    var E = engine();
    if (!E) {
      throw new Error("QAPI.setED(" + name + "): Qualtrics.SurveyEngine is not available");
    }
    var v = (value === null || value === undefined) ? "" : String(value);
    if (typeof E.setJSEmbeddedData === "function") {
      E.setJSEmbeddedData(name, v);
      return;
    }
    if (typeof E.setEmbeddedData === "function") {
      E.setEmbeddedData(name, v);
      return;
    }
    throw new Error("QAPI.setED(" + name + "): no embedded-data setter on " +
                    "Qualtrics.SurveyEngine — the value would have been lost");
  }

  /* Read one embedded-data field back, with a fallback. Never pipe ${e://...}
   * into JS to do this: piped text is resolved server-side before the page is
   * sent, so it is stale, and a quote in the value is a syntax error. */
  function getED(name, fallback) {
    var E = engine();
    var raw = (E && typeof E.getJSEmbeddedData === "function")
      ? E.getJSEmbeddedData(name) : null;
    return (raw === null || raw === undefined || raw === "") ? fallback : raw;
  }

  /* Register a page-submit handler — where all data writes belong. Data written
   * in addOnUnload races the page POST and is routinely lost. */
  function onSubmit(fn) {
    var E = engine();
    if (!E || typeof E.addOnPageSubmit !== "function") {
      throw new Error("QAPI.onSubmit: Qualtrics.SurveyEngine.addOnPageSubmit is " +
                      "unavailable — results would never be written");
    }
    E.addOnPageSubmit(fn);
  }

  root.QAPI = { setED: setED, getED: getED, onSubmit: onSubmit };
})(typeof window !== "undefined" ? window : this);

/* ============================================================================
 * qualtrics_api.js — the ONLY place the task engines touch Qualtrics.
 *
 * Why this file exists
 * --------------------
 * The embedded-data setters and the page-submit hook live on the STATIC
 * `Qualtrics.SurveyEngine` namespace, NOT on the question instance that
 * `this` refers to inside addOnload/addOnReady. Calling them on the instance
 * (`q.setJSEmbeddedData(...)`) throws, and every recorded value is lost — the
 * task renders and plays perfectly, then writes nothing. Guarding the call
 * (`if (q.addOnPageSubmit) ...`) makes that failure silent, which is worse.
 *
 * So: engines call QAPI.setED / QAPI.onSubmit. Those resolve the real static
 * API at call time and THROW LOUDLY if it is not there. A missing API must
 * break the local harness and the Qualtrics preview, not quietly drop data.
 *
 * Layout compatibility
 * --------------------
 * `setJSEmbeddedData` is the current setter and the only one that works in the
 * New Survey Taking Experience; `setEmbeddedData` is deprecated but still works
 * in the legacy layout. We prefer the former and fall back to the latter, so
 * the same code is correct under either layout.
 *
 * Values are coerced to String, and null/undefined become "" rather than the
 * literal text "null"/"undefined".
 * ==========================================================================*/
(function (root) {
  "use strict";

  function engine() {
    return root.Qualtrics && root.Qualtrics.SurveyEngine;
  }

  /* Write one embedded-data field. Throws if no setter is available. */
  function setED(name, value) {
    var E = engine();
    if (!E) {
      throw new Error("QAPI.setED(" + name + "): Qualtrics.SurveyEngine is not available");
    }
    var v = (value === null || value === undefined) ? "" : String(value);
    if (typeof E.setJSEmbeddedData === "function") {
      E.setJSEmbeddedData(name, v);
      return;
    }
    if (typeof E.setEmbeddedData === "function") {
      E.setEmbeddedData(name, v);
      return;
    }
    throw new Error("QAPI.setED(" + name + "): no embedded-data setter on " +
                    "Qualtrics.SurveyEngine — the value would have been lost");
  }

  /* Read one embedded-data field back, with a fallback. Never pipe ${e://...}
   * into JS to do this: piped text is resolved server-side before the page is
   * sent, so it is stale, and a quote in the value is a syntax error. */
  function getED(name, fallback) {
    var E = engine();
    var raw = (E && typeof E.getJSEmbeddedData === "function")
      ? E.getJSEmbeddedData(name) : null;
    return (raw === null || raw === undefined || raw === "") ? fallback : raw;
  }

  /* Register a page-submit handler — where all data writes belong. Data written
   * in addOnUnload races the page POST and is routinely lost. */
  function onSubmit(fn) {
    var E = engine();
    if (!E || typeof E.addOnPageSubmit !== "function") {
      throw new Error("QAPI.onSubmit: Qualtrics.SurveyEngine.addOnPageSubmit is " +
                      "unavailable — results would never be written");
    }
    E.addOnPageSubmit(fn);
  }

  root.QAPI = { setED: setED, getED: getED, onSubmit: onSubmit };
})(typeof window !== "undefined" ? window : this);
